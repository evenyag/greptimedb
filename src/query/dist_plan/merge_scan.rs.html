70fn acquire_remote_dyn_filter_registry_lease(
71    context: &TaskContext,
72    query_ctx: &QueryContextRef,
73    captured_dyn_filters: &[CapturedDynFilter],
74) -> Option<RemoteDynFilterRegistryLease> {
75    if captured_dyn_filters.is_empty() {
76        return None;
77    }
78
79    let query_id = query_ctx.remote_query_id_value()?;
80    let query_engine_state = query_engine_state_from_task_context(context)?;
81    Some(
82        query_engine_state
83            .dyn_filter_registry_manager()
84            .acquire_lease(query_id),
85    )
86}

88fn query_context_for_remote_dyn_filter_region(
89    query_ctx: &QueryContextRef,
90    region_id: RegionId,
91    remote_dyn_filter_registry_lease: Option<&RemoteDynFilterRegistryLease>,
92    captured_dyn_filters: &[CapturedDynFilter],
93) -> session::context::QueryContext {
94    if let Some(remote_dyn_filter_registry_lease) = remote_dyn_filter_registry_lease {
95        register_dyn_filters_for_region(
96            remote_dyn_filter_registry_lease.registry(),
97            region_id,
98            captured_dyn_filters,
99        );
100    }
101
102    query_context_with_initial_dyn_filter_regs(query_ctx, region_id, captured_dyn_filters)
103}

230impl MergeScanExec {
231    #[allow(clippy::too_many_arguments)]
232    pub fn new(
233        session_state: &SessionState,
234        table: TableName,
235        regions: Vec<RegionId>,
236        plan: LogicalPlan,
237        arrow_schema: &ArrowSchema,
238        region_query_handler: RegionQueryHandlerRef,
239        query_ctx: QueryContextRef,
240        target_partition: usize,
241        partition_cols: AliasMapping,
242        remote_dyn_filter_producer_id: Option<RemoteDynFilterProducerId>,
243    ) -> Result<Self> {
244        // TODO(CookiePieWw): Initially we removed the metadata from the schema in #2000, but we have to
245        // keep it for #4619 to identify json type in src/datatypes/src/schema/column_schema.rs.
246        // Reconsider if it's possible to remove it.
247        let arrow_schema = Arc::new(arrow_schema.clone());
248
249        // States the output ordering of the plan.
250        //
251        // When the input plan is a sort, we can use the sort ordering as the output ordering
252        // if the target partition is greater than the number of regions, which means we won't
253        // break the ordering on merging (of MergeScan).
254        //
255        // Otherwise, we need to use the default ordering.
256        let eq_properties = if let LogicalPlan::Sort(sort) = &plan
257            && target_partition >= regions.len()
258        {
259            let lex_ordering = sort
260                .expr
261                .iter()
262                .map(|sort_expr| {
263                    let physical_expr = session_state
264                        .create_physical_expr(sort_expr.expr.clone(), plan.schema())?;
265                    Ok(PhysicalSortExpr::new(
266                        physical_expr,
267                        SortOptions {
268                            descending: !sort_expr.asc,
269                            nulls_first: sort_expr.nulls_first,
270                        },
271                    ))
272                })
273                .collect::<Result<Vec<_>>>()?;
274            EquivalenceProperties::new_with_orderings(arrow_schema.clone(), vec![lex_ordering])
275        } else {
276            EquivalenceProperties::new(arrow_schema.clone())
277        };
278
279        let partition_exprs = partition_cols
280            .iter()
281            .filter_map(|col| {
282                if let Some(first_alias) = col.1.first() {
283                    session_state
284                        .create_physical_expr(
285                            Expr::Column(ColumnExpr::new_unqualified(
286                                first_alias.name().to_string(),
287                            )),
288                            plan.schema(),
289                        )
290                        .ok()
291                } else {
292                    None
293                }
294            })
295            .collect();
296        let partitioning = Partitioning::Hash(partition_exprs, target_partition);
297
298        let properties = Arc::new(PlanProperties::new(
299            eq_properties,
300            partitioning,
301            EmissionType::Incremental,
302            Boundedness::Bounded,
303        ));
304        Ok(Self {
305            table,
306            regions,
307            plan,
308            arrow_schema,
309            region_query_handler,
310            metric: ExecutionPlanMetricsSet::new(),
311            sub_stage_metrics: Arc::default(),
312            partition_metrics: Arc::default(),
313            properties,
314            query_ctx,
315            remote_dyn_filter_producer_id,
316            captured_remote_dyn_filters: Arc::default(),
317            target_partition,
318            partition_cols,
319        })
320    }

322    pub fn to_stream(
323        &self,
324        context: Arc<TaskContext>,
325        partition: usize,
326    ) -> Result<SendableRecordBatchStream> {
327        // prepare states to move
328        let regions = self.regions.clone();
329        let region_query_handler = self.region_query_handler.clone();
330        let metric = MergeScanMetric::new(&self.metric);
331        let arrow_schema = self.arrow_schema.clone();
332        let query_ctx = self.query_ctx.clone();
333        let sub_stage_metrics_moved = self.sub_stage_metrics.clone();
334        let partition_metrics_moved = self.partition_metrics.clone();
335        let plan = self.plan.clone();
336        let target_partition = self.target_partition;
337        let captured_remote_dyn_filters = self.captured_remote_dyn_filters();
338        let dbname = context.task_id().unwrap_or_default();
339        let tracing_context = TracingContext::from_json(context.session_id().as_str());
340        let current_channel = self.query_ctx.channel();
341        let read_preference = self.query_ctx.read_preference();
342        let explain_verbose = self.query_ctx.explain_verbose();
343        let remote_dyn_filter_registry_lease = acquire_remote_dyn_filter_registry_lease(
344            context.as_ref(),
345            &query_ctx,
346            &captured_remote_dyn_filters,
347        );
348
349        let stream = Box::pin(stream!({
350            let remote_dyn_filter_registry_lease = remote_dyn_filter_registry_lease;
351            // only report metrics once for each MergeScan
352            if partition == 0 {
353                MERGE_SCAN_REGIONS.observe(regions.len() as f64);
354            }
355
356            let _finish_timer = metric.finish_time().timer();
357            let mut ready_timer = metric.ready_time().timer();
358            let mut first_consume_timer = Some(metric.first_consume_time().timer());
359
360            for region_id in regions
361                .iter()
362                .skip(partition)
363                .step_by(target_partition)
364                .copied()
365            {
366                let region_span = tracing_context.attach(tracing::info_span!(
367                    parent: &Span::current(),
368                    "merge_scan_region",
369                    region_id = %region_id,
370                    partition = partition
371                ));
372                let region_query_ctx = query_context_for_remote_dyn_filter_region(
373                    &query_ctx,
374                    region_id,
375                    remote_dyn_filter_registry_lease.as_ref(),
376                    &captured_remote_dyn_filters,
377                );
378                let request = QueryRequest {
379                    header: Some(RegionRequestHeader {
380                        tracing_context: tracing_context.to_w3c(),
381                        dbname: dbname.clone(),
382                        query_context: Some((&region_query_ctx).into()),
383                    }),
384                    region_id,
385                    plan: plan.clone(),
386                };
387                let region_start = Instant::now();
388                let do_get_start = Instant::now();
389
390                if explain_verbose {
391                    common_telemetry::info!(
392                        "Merge scan one region, partition: {}, region_id: {}",
393                        partition,
394                        region_id
395                    );
396                }
397
398                let mut stream = region_query_handler
399                    .do_get(read_preference, request)
400                    .instrument(region_span.clone())
401                    .await
402                    .map_err(|e| {
403                        MERGE_SCAN_ERRORS_TOTAL.inc();
404                        DataFusionError::External(Box::new(e))
405                    })?;
406                let do_get_cost = do_get_start.elapsed();
407
408                if let Some(remote_dyn_filter_registry_lease) =
409                    remote_dyn_filter_registry_lease.as_ref()
410                {
411                    remote_dyn_filter_registry_lease
412                        .ensure_fanout_task(region_query_handler.clone());
413                }
414
415                ready_timer.stop();
416
417                let mut poll_duration = Duration::ZERO;
418                let mut poll_timer = Instant::now();
419                while let Some(batch) = stream.next().instrument(region_span.clone()).await {
420                    let poll_elapsed = poll_timer.elapsed();
421                    poll_duration += poll_elapsed;
422
423                    let batch = batch.map_err(|e| DataFusionError::External(Box::new(e)))?;
424                    let batch = patch_batch_timezone(
425                        arrow_schema.clone(),
426                        batch.into_df_record_batch().columns().to_vec(),
427                    )?;
428                    metric.record_output_batch_rows(batch.num_rows());
429                    if let Some(mut first_consume_timer) = first_consume_timer.take() {
430                        first_consume_timer.stop();
431                    }
432
433                    if let Some(metrics) = stream.metrics() {
434                        let mut sub_stage_metrics = sub_stage_metrics_moved.lock().unwrap();
435                        sub_stage_metrics.insert(region_id, metrics);
436                    }
437
438                    yield Ok(batch);
439                    // reset poll timer
440                    poll_timer = Instant::now();
441                }
442                let total_cost = region_start.elapsed();
443
444                // Record region metrics and push to global partition_metrics
445                let region_metrics = RegionMetrics {
446                    region_id,
447                    poll_duration,
448                    do_get_cost,
449                    total_cost,
450                };
451
452                // Push RegionMetrics to global partition_metrics immediately after scanning this region
453                {
454                    let mut partition_metrics_guard = partition_metrics_moved.lock().unwrap();
455                    let partition_metrics = partition_metrics_guard
456                        .entry(partition)
457                        .or_insert_with(|| PartitionMetrics::new(partition, explain_verbose));
458                    partition_metrics.add_region_metrics(region_metrics);
459                }
460
461                if explain_verbose {
462                    common_telemetry::info!(
463                        "Merge scan finish one region, partition: {}, region_id: {}, poll_duration: {:?}, first_consume: {}, do_get_cost: {:?}",
464                        partition,
465                        region_id,
466                        poll_duration,
467                        metric.first_consume_time(),
468                        do_get_cost
469                    );
470                }
471
472                // process metrics after all data is drained.
473                if let Some(metrics) = stream.metrics() {
474                    let (c, s) = parse_catalog_and_schema_from_db_string(&dbname);
475                    let value = read_meter!(
476                        c,
477                        s,
478                        ReadItem {
479                            cpu_time: metrics.elapsed_compute as u64,
480                            table_scan: metrics.memory_usage as u64
481                        },
482                        current_channel as u8
483                    );
484                    metric.record_greptime_exec_cost(value as usize);
485
486                    // record metrics from sub sgates
487                    let mut sub_stage_metrics = sub_stage_metrics_moved.lock().unwrap();
488                    sub_stage_metrics.insert(region_id, metrics);
489                }
490
491                MERGE_SCAN_POLL_ELAPSED.observe(poll_duration.as_secs_f64());
492            }
493
494            // Finish partition metrics and log results
495            {
496                let mut partition_metrics_guard = partition_metrics_moved.lock().unwrap();
497                if let Some(partition_metrics) = partition_metrics_guard.get_mut(&partition) {
498                    partition_metrics.finish();
499                }
500            }
501        }));
502
503        Ok(Box::pin(RecordBatchStreamAdapter::new(
504            self.arrow_schema.clone(),
505            stream,
506        )))
507    }

509    pub fn try_with_new_distribution(&self, distribution: Distribution) -> Option<Self> {
510        let Distribution::HashPartitioned(hash_exprs) = distribution else {
511            // not applicable
512            return None;
513        };
514
515        if let Partitioning::Hash(curr_dist, _) = &self.properties.partitioning
516            && curr_dist == &hash_exprs
517        {
518            // No need to change the distribution
519            return None;
520        }
521
522        let all_partition_col_aliases: HashSet<_> = self
523            .partition_cols
524            .values()
525            .flat_map(|aliases| aliases.iter().map(|c| c.name()))
526            .collect();
527        let overlaps: Vec<_> = hash_exprs
528            .iter()
529            .filter(|expr| {
530                expr.as_any()
531                    .downcast_ref::<Column>()
532                    .is_some_and(|col_expr| all_partition_col_aliases.contains(col_expr.name()))
533            })
534            .cloned()
535            .collect();
536
537        if overlaps.is_empty() {
538            return None;
539        }
540
541        Some(Self {
542            table: self.table.clone(),
543            regions: self.regions.clone(),
544            plan: self.plan.clone(),
545            arrow_schema: self.arrow_schema.clone(),
546            region_query_handler: self.region_query_handler.clone(),
547            metric: self.metric.clone(),
548            properties: Arc::new(PlanProperties::new(
549                self.properties.eq_properties.clone(),
550                Partitioning::Hash(overlaps, self.target_partition),
551                self.properties.emission_type,
552                self.properties.boundedness,
553            )),
554            sub_stage_metrics: self.sub_stage_metrics.clone(),
555            partition_metrics: self.partition_metrics.clone(),
556            query_ctx: self.query_ctx.clone(),
557            remote_dyn_filter_producer_id: self.remote_dyn_filter_producer_id,
558            captured_remote_dyn_filters: self.captured_remote_dyn_filters.clone(),
559            target_partition: self.target_partition,
560            partition_cols: self.partition_cols.clone(),
561        })
562    }

581    pub fn is_flow_sink_scan(&self) -> bool {
582        let Some(sink_table_id) =
583            FlowQueryExtensions::parse_flow_extensions(&self.query_ctx.extensions())
584                .ok()
585                .flatten()
586                .and_then(|extensions| extensions.sink_table_id)
587        else {
588            return false;
589        };
590
591        !self.regions.is_empty()
592            && self
593                .regions
594                .iter()
595                .all(|region_id| region_id.table_id() == sink_table_id)
596    }

623/// Metrics for a region of a partition.
624#[derive(Debug, Clone)]
625struct RegionMetrics {
626    region_id: RegionId,
627    poll_duration: Duration,
628    do_get_cost: Duration,
629    /// Total cost to scan the region.
630    total_cost: Duration,
631}
632
633/// Metrics for a partition of a MergeScanExec.
634#[derive(Debug, Clone)]
635struct PartitionMetrics {
636    partition: usize,
637    region_metrics: Vec<RegionMetrics>,
638    total_poll_duration: Duration,
639    total_do_get_cost: Duration,
640    total_regions: usize,
641    explain_verbose: bool,
642    finished: bool,
643}
644
645impl PartitionMetrics {
646    fn new(partition: usize, explain_verbose: bool) -> Self {
647        Self {
648            partition,
649            region_metrics: Vec::new(),
650            total_poll_duration: Duration::ZERO,
651            total_do_get_cost: Duration::ZERO,
652            total_regions: 0,
653            explain_verbose,
654            finished: false,
655        }
656    }
657
658    fn add_region_metrics(&mut self, region_metrics: RegionMetrics) {
659        self.total_poll_duration += region_metrics.poll_duration;
660        self.total_do_get_cost += region_metrics.do_get_cost;
661        self.total_regions += 1;
662        self.region_metrics.push(region_metrics);
663    }
664
665    /// Finish the partition metrics and log the results.
666    fn finish(&mut self) {
667        if self.finished {
668            return;
669        }
670        self.finished = true;
671        self.log_metrics();
672    }
673
674    /// Log partition metrics based on explain_verbose level.
675    fn log_metrics(&self) {
676        if self.explain_verbose {
677            common_telemetry::info!(
678                "MergeScan partition {} finished: {} regions, total_poll_duration: {:?}, total_do_get_cost: {:?}",
679                self.partition,
680                self.total_regions,
681                self.total_poll_duration,
682                self.total_do_get_cost
683            );
684        } else {
685            common_telemetry::debug!(
686                "MergeScan partition {} finished: {} regions, total_poll_duration: {:?}, total_do_get_cost: {:?}",
687                self.partition,
688                self.total_regions,
689                self.total_poll_duration,
690                self.total_do_get_cost
691            );
692        }
693    }

704impl ExecutionPlan for MergeScanExec {
705    fn as_any(&self) -> &dyn Any {
706        self
707    }
708
709    fn schema(&self) -> ArrowSchemaRef {
710        self.arrow_schema.clone()
711    }
712
713    fn properties(&self) -> &Arc<PlanProperties> {
714        &self.properties
715    }
716
717    fn children(&self) -> Vec<&Arc<dyn ExecutionPlan>> {
718        vec![]
719    }
720
721    // DataFusion will swap children unconditionally.
722    // But since this node is leaf node, it's safe to just return self.
723    fn with_new_children(
724        self: Arc<Self>,
725        _children: Vec<Arc<dyn ExecutionPlan>>,
726    ) -> Result<Arc<dyn ExecutionPlan>> {
727        Ok(self.clone())
728    }
729
730    fn handle_child_pushdown_result(
731        &self,
732        _phase: FilterPushdownPhase,
733        child_pushdown_result: ChildPushdownResult,
734        _config: &datafusion::config::ConfigOptions,
735    ) -> Result<FilterPushdownPropagation<Arc<dyn ExecutionPlan>>> {
736        let parent_filters = child_pushdown_result
737            .parent_filters
738            .into_iter()
739            .map(|filter| filter.filter)
740            .collect::<Vec<_>>();
741        let Some(remote_dyn_filter_producer_id) = self.remote_dyn_filter_producer_id else {
742            // Missing RDF identity disables only RDF, not normal execution.
743            common_telemetry::warn!(
744                "MergeScan remote dynamic filter producer id is not assigned; skipping remote dynamic filter pushdown"
745            );
746            self.captured_remote_dyn_filters.lock().unwrap().clear();
747            let new_self = Arc::new(self.clone());
748
749            return Ok(FilterPushdownPropagation {
750                filters: parent_filters.into_iter().map(|_| PushedDown::No).collect(),
751                updated_node: Some(new_self),
752            });
753        };
754        let remote_dyn_filter_pushdown =
755            capture_remote_dyn_filters_for_pushdown(remote_dyn_filter_producer_id, parent_filters);
756        *self.captured_remote_dyn_filters.lock().unwrap() =
757            remote_dyn_filter_pushdown.captured_dyn_filters;
758        let new_self = Arc::new(self.clone());
759
760        Ok(FilterPushdownPropagation {
761            filters: remote_dyn_filter_pushdown
762                .pushed_down
763                .into_iter()
764                .map(|_pushdown_ready| {
765                    // TODO(discord9): Return `PushedDown::Yes` after datanodes consume RDF
766                    // registrations and pending updates. Until then, keep the parent-side filter.
767                    PushedDown::No
768                })
769                .collect(),
770            updated_node: Some(new_self),
771        })
772    }
773
774    fn execute(
775        &self,
776        partition: usize,
777        context: Arc<TaskContext>,
778    ) -> Result<SendableRecordBatchStream> {
779        self.to_stream(context, partition)
780    }
781
782    fn metrics(&self) -> Option<MetricsSet> {
783        Some(self.metric.clone_inner())
784    }
785
786    fn name(&self) -> &str {
787        "MergeScanExec"
788    }
789}

792    fn fmt_as(&self, t: DisplayFormatType, f: &mut std::fmt::Formatter) -> std::fmt::Result {
793        write!(f, "MergeScanExec: peers=[")?;
794        for region_id in self.regions.iter() {
795            write!(f, "{}, ", region_id)?;
796        }
797        write!(f, "]")?;
798
799        if matches!(t, DisplayFormatType::Verbose) {
800            let partition_metrics = self.partition_metrics();
801            if !partition_metrics.is_empty() {
802                write!(f, ", metrics={{")?;
803                for (i, pm) in partition_metrics.iter().enumerate() {
804                    if i > 0 {
805                        write!(f, ", ")?;
806                    }
807                    write!(
808                        f,
809                        "\"partition_{}\":{{\"regions\":{},\"total_poll_duration\":\"{:?}\",\"total_do_get_cost\":\"{:?}\",\"region_metrics\":[",
810                        pm.partition,
811                        pm.total_regions,
812                        pm.total_poll_duration,
813                        pm.total_do_get_cost
814                    )?;
815                    for (j, rm) in pm.region_metrics.iter().enumerate() {
816                        if j > 0 {
817                            write!(f, ",")?;
818                        }
819                        write!(
820                            f,
821                            "{{\"region_id\":\"{}\",\"poll_duration\":\"{:?}\",\"do_get_cost\":\"{:?}\",\"total_cost\":\"{:?}\"}}",
822                            rm.region_id, rm.poll_duration, rm.do_get_cost, rm.total_cost
823                        )?;
824                    }
825                    write!(f, "]}}")?;
826                }
827                write!(f, "}}")?;
828            }
829        }
830
831        Ok(())
832    }
